================================================================================
                          TIPOS PRINCIPAIS
================================================================================

CodeGeneratorIntent
├─ intent: string (obrigatório)
├─ segment: string (obrigatório)
├─ mantaCode?: string
├─ tier?: "Haiku" | "Sonnet" | "Opus"
├─ keywords?: string[]
├─ userEmail?: string
└─ projectRoot?: string

CodeGeneratorOutput
├─ status: "success" | "partial" | "failed"
├─ artifacts: GeneratedArtifact[]
├─ createdFiles: string[]
├─ branchName: string
├─ commitHash: string
├─ errors: string[]
├─ warnings: string[]
├─ executionTimeMs: number
└─ conversationLog?: ConversationMessage[]

GeneratedArtifact
├─ filename: string
├─ filepath: string
├─ content: string
├─ type: "agent-md" | "test-cases" | "documentation" | "keywords-json"
├─ validated: boolean
└─ validationErrors?: string[]

================================================================================
                            USO BÁSICO
================================================================================

import { CodeGenerator, type CodeGeneratorIntent } from "./services";

const generator = new CodeGenerator();

const result = await generator.generateCode({
  intent: "Criar agente para ETA e ETE com cálculos de demanda",
  segment: "Saneamento",
  mantaCode: "Manta 03-S8",
  tier: "Sonnet",
  keywords: ["eta", "ete", "adutora"],
  userEmail: "mneves@mantaassociados.com",
});

if (result.status === "success") {
  console.log(`✓ Branch: ${result.branchName}`);
  console.log(`✓ Commit: ${result.commitHash}`);
  console.log(`✓ Artefatos: ${result.artifacts.length}`);
}
